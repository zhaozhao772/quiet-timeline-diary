function openIDB(){
  return new Promise((resolve, reject)=>{
    if(!window.indexedDB){ reject(new Error("no idb")); return; }
    let req;
    try{ req = indexedDB.open(IDB_NAME, IDB_VERSION); }
    catch(e){ reject(e); return; }
    req.onupgradeneeded = (event)=>{
      const db = req.result;
      /* 铁律：表不存在才创建；已有表（含老用户的 records）一律不碰，老数据天然保留 */
      if(!db.objectStoreNames.contains(IDB_STORE)) db.createObjectStore(IDB_STORE, {keyPath:"id"});
      if(!db.objectStoreNames.contains(IDB_STORE_SETTINGS)) db.createObjectStore(IDB_STORE_SETTINGS, {keyPath:"key"});
      if(!db.objectStoreNames.contains(IDB_STORE_BACKUPS)) db.createObjectStore(IDB_STORE_BACKUPS, {keyPath:"id"});
    };
    req.onsuccess = ()=> resolve(req.result);
    req.onerror = ()=> reject(req.error);
    req.onblocked = ()=> reject(new Error("blocked"));
    setTimeout(()=> reject(new Error("timeout")), 2500);
  });
}

  return {

/* =========================================================
   数据保险箱：库内自动备份（backups_meta 表，保留最近 10 份）
   原则：大动作（导入 / 恢复 / 手动）前先留底，出事可一键回滚
========================================================= */
const BACKUP_REASONS = { manual:"手动备份", import:"导入前自动", "pre-restore":"恢复前自动", "pre-clear":"清空前自动" };
async function createBackup(reason){
  try{
    const now = Date.now();
    const id = `bk_${now}_${Math.random().toString(36).slice(2,6)}`;
    await idbPutIn(IDB_STORE_BACKUPS, {
      id, createdAt: now,
      reason: BACKUP_REASONS[reason] || reason || "自动备份",
      recordCount: state.records.length,
      folderCount: state.folders.length,
      records: state.records,   /* 全量快照（含回收站记录，回滚才完整） */
      folders: state.folders,
    });
    /* 滚动淘汰：只保留最近 BACKUP_KEEP 份 */
    const all = await listBackups();
    for(let i = BACKUP_KEEP; i < all.length; i++) await idbDelIn(IDB_STORE_BACKUPS, all[i].id);
    return id;
  }catch(e){ return null; }
}
async function listBackups(){
  const all = await idbAllIn(IDB_STORE_BACKUPS);
  return all.sort((a,b)=> b.createdAt - a.createdAt); /* 新的在前 */
}

async function restoreBackupById(id){
  const b = await idbGetIn(IDB_STORE_BACKUPS, id);
  if(!b || !Array.isArray(b.records)){ toast("这份备份读不出来了"); return false; }
  await createBackup("pre-restore");            /* 恢复前先备份当前，可再反悔 */
  await idbClear();
  state.folders = (b.folders || []).map(normalizeFolder).filter(Boolean);
  state.records = sortRecs(b.records.map(r=>normalizeRecord(r,"default")).filter(Boolean));
  if(!state.folders.some(f=>f.id === "default")){
    state.folders.push({ id:"default", name:"全部记录", createdAt: Date.now() });
  }
  state.records.forEach(r=>{ if(!getFolder(r.folderId)) r.folderId = "default"; });
  for(const r of state.records) idbPut(r);
  writeMirror(state.records, state.folders);
  renderHome();
  renderSidebar();
  return true;
}

function updateCarouselDots(){
  const nl = $("#note-list");
  if(!nl || !nl.children.length) return;
  const n = nl.children.length;
  const page = nl.scrollWidth / n;
  const idx = Math.min(n-1, Math.max(0, Math.round(nl.scrollLeft / page)));
  document.querySelectorAll("#carousel-dots .dot").forEach((d,i)=> d.classList.toggle("active", i===idx));

function scrollToCard(index, inFolder){

function saveEditor(){
  const title = $("#edit-title").value.trim();
  const added = $("#edit-content").value;
  if(!title && !added.trim()){ toast("写一点什么，再保存吧"); return; }

  const now = Date.now();
  if(state.editingId){
    /* 追加式保存 */
    const r = R().find(x=>x.id===state.editingId);
    if(!r) return;
    if(!added.trim()){ toast("本次还没有新增内容"); return; }
    r.title = title;  /* 标题可直接修改，不属于历史正文 */
    r.content = r.content + appendStamp(added, now);
    r.updatedAt = now;   /* 三时间戳：更新时间 */
    idbPut(r);
    toast("已追加保存");
  }else{
    /* 新建 */
    if(!added.trim()){ toast("写一点什么，再保存吧"); return; }
    const rec = {id:newId(), title, content:added, createdAt:now, date:now, folderId: state.currentFolderId || "default", updatedAt:now, deletedAt:null};
    state.records.push(rec);
    idbPut(rec);
    toast("已保存");
  }
  state.records = sortRecs(state.records);
  writeMirror(state.records, state.folders);
  renderSidebar();
  if(state.returnTo.view === "folder"){
    showFolder(state.returnTo.folderId || "default");
  }else{
    showHome(false);
  }
  if(state.editingId){
    setTimeout(()=> locateRecord(state.editingId), 350);
    state.editingId = null;
  }else{
    const newest = R()[R().length-1];
    if(newest) setTimeout(()=> locateRecord(newest.id), 350);
  }
}

function normalizeImported(raw){
  if(!raw || typeof raw !== "object") return null;
  const isV2 = Array.isArray(raw);
  const isV3 = raw && Array.isArray(raw.records);
  const recList = isV2 ? raw : (isV3 ? raw.records : null);
  if(!recList || !Array.isArray(recList)) return null;
  const folderList = isV3 && Array.isArray(raw.folders) ? raw.folders : null;
  const folders = folderList ? folderList.map(normalizeFolder).filter(Boolean) : [];
  const records = recList.map(r=>normalizeRecord(r)).filter(Boolean);
  return { folders, records };
}
function handleImport(raw){
  const backup = normalizeImported(raw);
  if(!backup || !backup.records.length){ toast("这个文件里没有可用的记录"); return; }
  /* 若备份无 folders，生成一个默认文件夹并把所有记录归到里面 */
  if(!backup.folders.length){
    backup.folders.push({ id: "default", name: "导入的记录", createdAt: Date.now() });
    backup.records.forEach(r=> r.folderId = "default");
  }
  const local = R();
  const keyOf = r => `${r.createdAt}|${r.title}|${r.content}`;
  const localKeys = new Set(local.map(keyOf));
  const localIds = new Set(local.map(r=>r.id));
  const same = backup.records.filter(r=>localKeys.has(keyOf(r))).length;
  const fresh = backup.records.length - same;
  showModal({
    title: "导入备份",
    body: `备份中共 ${backup.records.length} 篇记录，\n其中 ${same} 篇与当前记录相同，${fresh} 篇是新的。\n\n合并导入：把新记录加入现有记录；\n覆盖当前数据：清空现有 ${local.length} 篇，换成备份内容。`,
    buttons: [
      {text:"合并导入（推荐）", cls:"mbtn-main", onClick: async ()=>{
        await createBackup("import");   /* 导入前自动留底，可回滚 */
        /* 合并 folders：避免 ID/名称冲突 */
        const folderMap = new Map(state.folders.map(f=>[f.id, f]));
        backup.folders.forEach(f=>{ if(!folderMap.has(f.id)) { folderMap.set(f.id, f); state.folders.push(f); } });
        let added = 0;
        for(const r of backup.records){
          if(localKeys.has(keyOf(r))) continue;
          const rec = {...r};
          if(localIds.has(rec.id)) rec.id = newId();
          if(!folderMap.has(rec.folderId)) rec.folderId = "default";
          state.records.push(rec);
          idbPut(rec);
          localIds.add(rec.id);
          added++;
        }
        state.records = sortRecs(state.records);
        writeMirror(state.records, state.folders);
        renderHome();
        if(state.view === "folder") renderFolder(state.currentFolderId);
        renderSidebar();
        toast(added ? `已合并导入 ${added} 篇记录` : "备份中的记录都已存在");
      }},
      {text:"覆盖当前数据", cls:"mbtn-danger", onClick: ()=>{
        showModal({
          title: "确定覆盖？",
          body: `覆盖会删除当前的 ${local.length} 篇记录，\n替换为备份中的 ${backup.records.length} 篇。`,
          buttons: [
            {text:"再想想", cls:"mbtn-ghost"},
            {text:"确定覆盖", cls:"mbtn-danger", onClick: async ()=>{
              await createBackup("import");   /* 覆盖前自动留底当前全部数据 */
              await idbClear();
              state.folders = backup.folders;
              state.records = sortRecs(backup.records);
              state.records.forEach(idbPut);
              writeMirror(state.records, state.folders);
              renderHome();
              renderSidebar();
              toast(`已恢复 ${state.records.length} 篇记录`);
            }},
          ],
        });
      }},
      {text:"取消", cls:"mbtn-ghost"},
    ],
  });
}
$("#import-file").addEventListener("change", e=>{
  const f = e.target.files && e.target.files[0];
  if(!f) return;
  const reader = new FileReader();
  reader.onload = ()=>{
    try{ handleImport(JSON.parse(reader.result)); }
    catch(err){ toast("无法读取这个备份文件"); }
  };
  reader.onerror = ()=> toast("无法读取这个备份文件");
  reader.readAsText(f, "utf-8");
  e.target.value = "";
});

async function renderStorageStatus(){
  const el = $("#storage-status");
  if(!el) return;
  let persisted = false;
  try{ persisted = !!(navigator.storage && await navigator.storage.persisted()); }catch(e){}
  let usedText = "";
  try{
    const est = await navigator.storage.estimate();
    if(est && est.usage != null) usedText = `，已用 ${(est.usage/1048576).toFixed(1)} MB`;
  }catch(e){}
  el.textContent = `存储保护：${persisted ? "已开启（降低被系统清理的风险）" : "未开启"}${usedText}`;
}

function showModal(opt){

  $("#modal-body").textContent = opt.body || "";
  const acts = $("#modal-acts");

      closeModal();
      if(b.onClick) b.onClick();
    });
    acts.appendChild(el);
  });

}

(async function init(){
  try{ idb = await openIDB(); }catch(e){ idb = null; }

  /* 0. 申请持久化存储：降低浏览器在存储压力下自动清理 IndexedDB 的概率 */
  try{ if(navigator.storage && navigator.storage.persist) await navigator.storage.persist(); }catch(e){}

  /* 1. 读取 records */
  let recs = null;
  if(idb) recs = await idbAll();
  let mirror = readMirror();
  if((!recs || !recs.length) && mirror && Array.isArray(mirror.records) && mirror.records.length){
    recs = mirror.records;
    if(idb) recs.forEach(idbPut);
  }

  /* 2. 读取 folders（旧版 mirror 没有 folders，需生成默认） */
  let folders = mirror && Array.isArray(mirror.folders) ? mirror.folders : [];

  /* 3. 首次使用：欢迎记录 + 默认文件夹 */
  if((!recs || !recs.length) && !localStorage.getItem(INIT_FLAG)){
    recs = welcomeRecords();
    folders = [{ id: "default", name: "全部记录", createdAt: Date.now() }];
    if(idb) recs.forEach(idbPut);
    try{ localStorage.setItem(INIT_FLAG, "1"); }catch(e){}
  }

  /* 4. 规范化数据 */
  state.records = sortRecs((recs || []).map(r=>normalizeRecord(r, "default")).filter(Boolean));
  state.folders = folders.map(normalizeFolder).filter(Boolean);

  /* 5. 数据迁移：确保默认文件夹存在，并把无主记录绑定到它 */
  if(!state.folders.some(f=>f.id === "default")){
    state.folders.push({ id: "default", name: "全部记录", createdAt: Date.now() });
  }
  state.records.forEach(r=>{
    if(!r.folderId || !getFolder(r.folderId)) r.folderId = "default";
  });

  /* 6. 持久化并渲染 */
  writeMirror(state.records, state.folders);
  renderHome();
  showView("home", "vin-u");
})();
